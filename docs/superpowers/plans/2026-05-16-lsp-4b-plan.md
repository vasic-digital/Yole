# LSP Capability Expansion — Implementation Plan (iter-62, Feature 4b of 5)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Extend iter-61's LSP integration with 3 capabilities (diagnostics rendered in editor across 3 surfaces, hover popups with sanitized Markdown, go-to-definition with in-place navigation + multi-result chooser).

**Architecture:** `DiagnosticsCache` (StateFlow single source of truth) consumed by 3 render paths (gutter dots, inline wavy underlines, Problems panel). `LspServerHost` extended with `hover()` + `definition()` suspend methods. `HoverMarkdownRenderer` parses LSP hover Markdown via Flexmark → Compose blocks. `EditorNavigationStack` backs device-back navigation after go-to-def.

**Tech Stack:** Kotlin Multiplatform 2.0.20, Compose Multiplatform 1.7.3, Eclipse LSP4J 1.0.0 (iter-61), Flexmark 0.64.8 (existing markdown preview dep), Tree-Sitter via bonede (iter-58), JUnit4 + `runBlocking<Unit>`, Robolectric for Android UI tests, Firebase App Distribution.

**Reference design spec:** `docs/superpowers/specs/2026-05-16-lsp-4b-design.md` (committed `8500f78d`).

```kotlin
// shared/src/commonMain/kotlin/digital/vasic/yole/lsp/DiagnosticsCache.kt
/*#######################################################
 * SPDX-FileCopyrightText: 2026 Milos Vasic
 * SPDX-License-Identifier: Apache-2.0
 * iter-62 Phase 1: DiagnosticsCache — single source of truth
 * for LSP-emitted diagnostics. Per-URI cap 1000.
 *
 * Populated by YoleLanguageClient.publishDiagnostics
 * (wired in Phase 2). Consumed by the 3 render surfaces
 * (Phase 5) via the states StateFlow.
 *
 * Cache semantic: LSP publishDiagnostics is full-state per URI
 * (server emits the complete current list). upsert replaces.
 *#######################################################*/
package digital.vasic.yole.lsp

import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update

class DiagnosticsCache {
    companion object {
        const val PER_URI_CAP: Int = 1000
    }

    private val _state = MutableStateFlow<Map<String, List<Diagnostic>>>(emptyMap())
    val states: StateFlow<Map<String, List<Diagnostic>>> = _state.asStateFlow()

    fun upsert(uri: String, diagnostics: List<Diagnostic>) {
        val capped = if (diagnostics.size > PER_URI_CAP) diagnostics.take(PER_URI_CAP) else diagnostics
        _state.update { current ->
            if (capped.isEmpty()) current - uri
            else current + (uri to capped)
        }
    }

    fun clear(uri: String) {
        _state.update { it - uri }
    }

    fun clearAll() {
        _state.value = emptyMap()
    }

    fun diagnosticsFor(uri: String): List<Diagnostic> = _state.value[uri].orEmpty()
}
```

```kotlin
        documentUri: String,
        line: Int,
        character: Int,
    ): HoverInfo? {
        val server = servers[langId] ?: return null
        return try {
            withTimeout(500L) {
                val params = org.eclipse.lsp4j.HoverParams(
                    org.eclipse.lsp4j.TextDocumentIdentifier(documentUri),
                    org.eclipse.lsp4j.Position(line, character),
                )
                val raw = server.languageServer.textDocumentService.hover(params)
                    .get(500, java.util.concurrent.TimeUnit.MILLISECONDS) ?: return@withTimeout null
                val contents = mapHoverContentsToMarkdown(raw.contents) ?: return@withTimeout null
                val range = raw.range?.let { mapLspRangeToIntRange(it, ::lineColToOffset.curry(documentUri)) }
                HoverInfo(contents, range)
            }
        } catch (e: kotlinx.coroutines.TimeoutCancellationException) {
            null
        } catch (e: kotlinx.coroutines.CancellationException) {
            throw e
        } catch (_: Throwable) {
            null
        }
    }

    private fun mapHoverContentsToMarkdown(
        contents: org.eclipse.lsp4j.jsonrpc.messages.Either<
            MutableList<org.eclipse.lsp4j.jsonrpc.messages.Either<String, org.eclipse.lsp4j.MarkedString>>,
            org.eclipse.lsp4j.MarkupContent
        >,
    ): String? = when {
        contents == null -> null
        contents.isRight -> contents.right.value
        contents.isLeft -> contents.left.joinToString("\n") { item ->
            if (item.isLeft) item.left
            else "```${item.right.language}\n${item.right.value}\n```"
        }
        else -> null
    }

    private fun mapLspRangeToIntRange(
        range: org.eclipse.lsp4j.Range,
        lineColToOffset: (Int, Int) -> Int,
    ): IntRange {
        val start = lineColToOffset(range.start.line, range.start.character)
        val end = lineColToOffset(range.end.line, range.end.character)
        return start..end
    }
```

Note: `lineColToOffset(uri, line, col)` requires the document text. For Phase 2, range mapping is best-effort — if the document isn't cached in the host, return `null` for the range (HoverInfo still useful without range). Phase 3 task 3.3 finalizes the mapping.

```kotlin
// shared/src/commonMain/kotlin/digital/vasic/yole/lsp/LspRangeMapping.kt
/*#######################################################
 * SPDX-FileCopyrightText: 2026 Milos Vasic
 * SPDX-License-Identifier: Apache-2.0
 * iter-62 Phase 3: Pure helper for LSP line/character → offset
 * conversion within a known document text.
 *
 * LSP uses 0-based line + 0-based char (UTF-16 code unit index per
 * the spec). Yole uses Kotlin String offsets (also 0-based UTF-16).
 * 1:1 conversion modulo line-by-line walk.
 *#######################################################*/
package digital.vasic.yole.lsp

object LspRangeMapping {
    fun lineColToOffset(text: String, line: Int, col: Int): Int {
        var currentLine = 0
        var lineStart = 0
        for (i in text.indices) {
            if (currentLine == line) {
                return (lineStart + col).coerceAtMost(text.length)
            }
            if (text[i] == '\n') {
                currentLine++
                lineStart = i + 1
            }
        }
        return text.length
    }
}
```

```kotlin
// shared/src/desktopMain/kotlin/digital/vasic/yole/lsp/HoverMarkdownRenderer.desktop.kt
/*#######################################################
 * SPDX-FileCopyrightText: 2026 Milos Vasic
 * SPDX-License-Identifier: Apache-2.0
 * iter-62 Phase 4: Desktop JVM actual using Flexmark.
 *
 * Walks the Flexmark Node tree emitting HoverBlock items.
 * Supported nodes:
 *   - Heading → HoverBlock.Heading
 *   - Paragraph → HoverBlock.Paragraph
 *   - FencedCodeBlock + IndentedCodeBlock → HoverBlock.CodeBlock
 *   - Code (inline) standalone at block-level → HoverBlock.InlineCodeSpan
 * Unsupported (rendered as FallbackText with raw chars):
 *   - Link, Image, Table, BulletList, OrderedList, BlockQuote, ThematicBreak
 *#######################################################*/
package digital.vasic.yole.lsp

import com.vladsch.flexmark.parser.Parser
import com.vladsch.flexmark.util.ast.Document
import com.vladsch.flexmark.util.ast.Node
import com.vladsch.flexmark.ast.FencedCodeBlock
import com.vladsch.flexmark.ast.Heading
import com.vladsch.flexmark.ast.IndentedCodeBlock
import com.vladsch.flexmark.ast.Paragraph

actual object HoverMarkdownRenderer {
    private val parser: Parser = Parser.builder().build()

    actual fun render(markdown: String): List<HoverBlock> {
        if (markdown.isEmpty()) return emptyList()
        val doc: Document = parser.parse(markdown)
        return buildList {
            var node: Node? = doc.firstChild
            while (node != null) {
                when (node) {
                    is Heading -> add(HoverBlock.Heading(node.level, node.text.toString()))
                    is Paragraph -> add(HoverBlock.Paragraph(node.chars.toString().trim()))
                    is FencedCodeBlock -> add(
                        HoverBlock.CodeBlock(
                            lang = node.info.toString().ifEmpty { null },
                            code = node.contentChars.toString(),
                        )
                    )
                    is IndentedCodeBlock -> add(HoverBlock.CodeBlock(lang = null, code = node.contentChars.toString()))
                    else -> add(HoverBlock.FallbackText(node.chars.toString().trim()))
                }
                node = node.next
            }
        }
    }
}
```
