data class TextEdit(val range: IntRange, val newText: String) {
    fun apply(text: String): String {
        val start = range.first.coerceIn(0, text.length)
        val end = (range.last + 1).coerceIn(start, text.length)
        return text.substring(0, start) + newText + text.substring(end)
    }
}

- [ ] **1.3: WorkspaceEditApplier** — `object WorkspaceEditApplier { fun apply(edit: WorkspaceEdit, sources: Map<String, String>): Map<String, String> }`. Applies edits per-file; preserves untouched URIs; conflict detection (overlapping ranges → throw ApplyConflict). Tests: 4 cases (single-file single-edit; multi-file; conflict; non-existent URI). Mutate: stub apply to return sources unchanged → 3 FAIL.

```kotlin
object WorkspaceEditApplier {
    class ApplyConflict(message: String) : RuntimeException(message)

    fun apply(edit: WorkspaceEdit, sources: Map<String, String>): Map<String, String> {
        if (edit.isEmpty) return sources
        val result = sources.toMutableMap()
        for ((uri, edits) in edit.changes) {
            if (edits.isEmpty()) continue
            // Validate non-overlapping
            val sorted = edits.sortedBy { it.range.first }
            for (i in 1 until sorted.size) {
                if (sorted[i - 1].range.last >= sorted[i].range.first) {
                    throw ApplyConflict("Overlapping edits in $uri")
                }
            }
            val source = result[uri] ?: continue  // honest: skip untracked URIs
            // Apply in reverse so offsets stay valid
            var modified = source
            for (e in sorted.reversed()) {
                modified = e.apply(modified)
            }
            result[uri] = modified
        }
        return result
    }
}
```
